% Regge curvature concentrated at a vertex: glue k unit equilateral triangles
% around one point.  Top row: the surface that results in 3-space.
% Bottom row: the same triangles unfolded flat, where the deficit is measured.
% Orthographic projection (x,y,z) |-> (x, sq*y + z), camera ~25 deg above the plane.
\begin{tikzpicture}[line join=round, line cap=round, font=\small]

\def\sq{0.42}      % foreshortening of the horizontal plane
\def\Q{1.8}        % scale of the 3-d panels

%=================== k = 5 : deficit > 0, the surface cones up ==============
\begin{scope}[shift={(0,0)}, scale=\Q]
  \def\rr{0.8507}\def\hh{0.5257}          % r = 1/(2 sin 36), h = sqrt(1-r^2)
  \foreach \i in {1,0,2,4,3} {            % painter's order, back to front
    \pgfmathsetmacro{\ta}{\i*72}
    \pgfmathsetmacro{\tb}{\ta+72}
    \pgfmathsetmacro{\sh}{45-22*sin(\ta+36)}
    \filldraw[fill=orange!\sh!white, draw=orange!55!black, line width=0.4pt]
      (0,\hh) -- ({\rr*cos(\ta)},{\sq*\rr*sin(\ta)})
              -- ({\rr*cos(\tb)},{\sq*\rr*sin(\tb)}) -- cycle;
  }
  \fill[orange!60!black] (0,\hh) circle (0.028);
  \draw[<-, orange!60!black, shorten <=1pt] (0,{\hh+0.06}) -- (0,{\hh+0.34});
\end{scope}
\node[align=center] at (0,-1.65) {$k=5$, cone \\ $K>0$};

%=================== k = 6 : deficit = 0, stays flat =======================
\begin{scope}[shift={(4.6,0)}, scale=\Q]
  \foreach \i in {0,...,5} {
    \pgfmathsetmacro{\ta}{\i*60}
    \pgfmathsetmacro{\tb}{\ta+60}
    \filldraw[fill=green!22!white, draw=green!45!black, line width=0.4pt]
      (0,0) -- ({cos(\ta)},{\sq*sin(\ta)}) -- ({cos(\tb)},{\sq*sin(\tb)}) -- cycle;
  }
  \fill[green!45!black] (0,0) circle (0.028);
\end{scope}
\node[align=center] at (4.6,-1.65) {$k=6$, flat \\ $K=0$};

%=================== k = 7 : deficit < 0, the surface buckles ==============
% rim vertices ride a three-fold wave, z = A cos(3 theta), radius sqrt(1-z^2),
% so every rim vertex stays at unit distance from the apex.
\begin{scope}[shift={(9.2,0)}, scale=\Q]
  \def\A{0.34}
  \foreach \i in {1,2,0,3,6,4,5} {        % painter's order, back to front
    \pgfmathsetmacro{\ta}{\i*51.42857}
    \pgfmathsetmacro{\tb}{\ta+51.42857}
    \pgfmathsetmacro{\za}{\A*cos(3*\ta)}
    \pgfmathsetmacro{\zb}{\A*cos(3*\tb)}
    \pgfmathsetmacro{\ra}{sqrt(1-\za*\za)}
    \pgfmathsetmacro{\rb}{sqrt(1-\zb*\zb)}
    \pgfmathsetmacro{\sh}{40+110*(\za+\zb)}
    \filldraw[fill=blue!\sh!white, draw=blue!55!black, line width=0.4pt]
      (0,0) -- ({\ra*cos(\ta)},{\sq*\ra*sin(\ta)+\za})
            -- ({\rb*cos(\tb)},{\sq*\rb*sin(\tb)+\zb}) -- cycle;
  }
  \fill[blue!55!black] (0,0) circle (0.028);
\end{scope}
\node[align=center] at (9.2,-1.65) {$k=7$, saddle \\ $K<0$};

%=================== bottom row: unfold and measure the deficit ============
\begin{scope}[shift={(0,-3.3)}]
  \foreach \i in {0,...,4} {
    \filldraw[fill=orange!20, draw=orange!55!black, line width=0.4pt]
      (0,0) -- ({\i*60}:1) -- ({(\i+1)*60}:1) -- cycle;
  }
  \draw[dashed, orange!70!black] (0,0) -- (300:1) arc (300:360:1) -- cycle;
  \draw[<->, orange!70!black] (300:0.55) arc (300:360:0.55);
  \fill[orange!60!black] (0,0) circle (0.025);
  \node[orange!60!black] at (330:1.42) {$\delta=+60^\circ$};
  \node[align=center] at (0,-1.4) {$5\times 60^\circ = 300^\circ$};
\end{scope}

\begin{scope}[shift={(4.6,-3.3)}]
  \foreach \i in {0,...,5} {
    \filldraw[fill=green!18, draw=green!45!black, line width=0.4pt]
      (0,0) -- ({\i*60}:1) -- ({(\i+1)*60}:1) -- cycle;
  }
  \fill[green!45!black] (0,0) circle (0.025);
  \node[green!45!black] at (330:1.42) {$\delta=0$};
  \node[align=center] at (0,-1.4) {$6\times 60^\circ = 360^\circ$};
\end{scope}

\begin{scope}[shift={(9.2,-3.3)}]
  \foreach \i in {0,...,5} {
    \filldraw[fill=blue!14, draw=blue!55!black, line width=0.4pt]
      (0,0) -- ({\i*60}:1) -- ({(\i+1)*60}:1) -- cycle;
  }
  % the seventh triangle has no room, it laps back over the first
  \begin{scope}[shift={(0.28,0.30)}]
    \filldraw[dashed, fill=blue!45, fill opacity=0.6, draw=blue!65!black,
              line width=0.4pt]
      (0,0) -- (0:1) -- (60:1) -- cycle;
  \end{scope}
  \draw[<->, blue!65!black] (0:0.55) arc (0:60:0.55);
  \fill[blue!55!black] (0,0) circle (0.025);
  \node[blue!65!black] at (330:1.42) {$\delta=-60^\circ$};
  \node[align=center] at (0,-1.4) {$7\times 60^\circ = 420^\circ$};
\end{scope}

\end{tikzpicture}
