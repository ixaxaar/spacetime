% Companion figure to Example (ex:run).
% Left: the simplicial set after the six events, composites in green with
% their witnessing 2-cells shaded. Right: the causal order as a Hasse
% diagram, with the causal past of g in blue and the incomparable event
% s3 in grey.
\begin{tikzpicture}[>=Stealth, thick, scale=1.0]

  % ---- left panel: the simplicial set ----
  % witnessing 2-cells first, so the edges sit on top
  \fill[green!18] (0,0) -- (1.8,0.9) -- (3.6,0) -- cycle;
  \fill[orange!20] (0,0) -- (3.6,0) -- (4.6,1.1) -- cycle;

  % vertices
  \fill (0,0) circle (1.8pt)   node[below left=2pt]  {\small $v_0$};
  \fill (1.8,0.9) circle (1.8pt) node[above=2pt]     {\small $v_1$};
  \fill (3.6,0) circle (1.8pt)  node[below=2pt]      {\small $v_2$};
  \fill[gray] (1.8,-1.1) circle (1.8pt) node[below=2pt, gray] {\small $v_3$};
  \fill (4.6,1.1) circle (1.8pt) node[above=2pt]     {\small $v_4$};

  % sprout edges
  \draw[->] (0.08,0.04) -- (1.7,0.86)  node[midway, above left=-1pt] {\small $s_1$};
  \draw[->] (1.9,0.84) -- (3.5,0.06)   node[midway, above right=-1pt] {\small $s_2$};
  \draw[->, gray] (0.06,-0.06) -- (1.72,-1.04) node[midway, below left=-1pt, gray] {\small $s_3$};
  \draw[->] (3.68,0.06) -- (4.52,1.04) node[midway, above right=-1pt] {\small $s_4$};

  % composite edges from the two fills
  \draw[->, green!55!black, very thick] (0.12,-0.08) -- (3.48,-0.08)
        node[midway, below] {\small $f$};
  \draw[->, orange!75!black, very thick] (0.1,0.12) -- (4.42,1.16)
        node[midway, above left=-2pt] {\small $g$};

  % ---- right panel: the causal order ----
  \begin{scope}[xshift=8.2cm]
    % nodes: past of g in blue, s3 in grey
    \fill[blue!70!black] (0,0) circle (2.4pt)    node[right=4pt, black] {\small $s_1$};
    \fill[blue!70!black] (0,1.1) circle (2.4pt)  node[right=4pt, black] {\small $s_2$};
    \fill[blue!70!black] (-0.8,2.2) circle (2.4pt) node[left=4pt, black] {\small $f$};
    \fill[blue!70!black] (0.8,2.2) circle (2.4pt)  node[right=4pt, black] {\small $s_4$};
    \fill[blue!70!black] (0,3.3) circle (2.4pt)  node[right=4pt, black] {\small $g$};
    \fill[gray] (1.8,0) circle (2.4pt)           node[right=4pt, gray] {\small $s_3$};

    % dependency arrows
    \draw[->, blue!70!black] (0,0.12) -- (0,0.98);
    \draw[->, blue!70!black] (-0.08,1.2) -- (-0.72,2.1);
    \draw[->, blue!70!black] (0.08,1.2) -- (0.72,2.1);
    \draw[->, blue!70!black] (-0.72,2.32) -- (-0.08,3.2);
    \draw[->, blue!70!black] (0.72,2.32) -- (0.08,3.2);

    \node[align=center] at (0.3,-1.0) {\small causal order,\\ \small past of $g$ in blue};
  \end{scope}

\end{tikzpicture}
